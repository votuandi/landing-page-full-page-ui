## Thiết kế
**`packages/plans/entitlements.ts`** (thêm dependency `zod` — đã chấp thuận trong AGENTS.md):
```ts
export const TRIAL_DAYS = 14;
export const ENTITLEMENTS_TAG = "platform:entitlements";
export const DISABLED_FEATURES_KEY = "features.disabled";
export const featureOverridesSchema; // record<Feature, { on: boolean; until?: ISO datetime; reason: string(min 1) }>
export const limitOverridesSchema;   // record<LimitKey, { add: int ≥ 0; until?: ISO datetime; reason: string(min 1) }>
export const disabledFeaturesSchema; // Feature[]
export type EntitlementSubject = { planId: string; addons: readonly string[]; status: string; createdAt: Date;
  featureOverrides?: unknown; limitOverrides?: unknown };
export type EntitlementContext = { disabled?: readonly string[]; now?: Date };
export function can(site, feature: string, ctx?): boolean;
export function limit(site, key: LimitKey, ctx?): number;          // gói + override `add` còn hạn
export function canUse(site, ctx?): (feature: string) => boolean;   // cho PageRenderer/SiteWidgets
export class EntitlementError extends Error { status: 403 | 413; code: Feature | LimitKey }
export function assertCan(site, feature, ctx?): void;              // ném EntitlementError 403
export function assertWithinLimit(site, key, used, adding, ctx?): void; // mediaGb tính theo byte → 413, còn lại → 403
```
- Thứ tự `can`: feature lạ → `false`; trong `disabled` → `false`; override hợp lệ, chưa hết hạn → `on`; ngược lại
  gói ∪ addons ∪ (trial còn hạn → toàn bộ premium). Sau cùng `featureDependencies` phải cùng `can` (quoteCart cần catalog).
- Override JSON parse từng mục (`safeParse` theo key); mục sai/không phải feature bị bỏ qua — dữ liệu xấu không mở khóa.
- Thông báo lỗi tiếng Việt lấy `featureLabels` / `limitLabels`.
